    "Hackers Hub" v5

		-- ==================== AUTOFARM ====================

		local Workspace = game:GetService("Workspace")

		local targetCoinName = "Coin_Server"
		local distanceFromCoin = 3 -- Отступ под монеткой (в стадах)
		local tweenSpeedStudsPerSec = 16 -- Скорость: 16 studs/sec

		local isFarming = false -- Флаг работы автофарма
		local activeTween = nil -- Переменная для хранения текущего твина

		-- Поиск ближайшей монетки Coin_Server
		local function getClosestCoin(character)
			local hrp = character:FindFirstChild("HumanoidRootPart")
			if not hrp then return nil end

			local closestCoin = nil
			local shortestDistance = math.huge

			for _, object in ipairs(Workspace:GetDescendants()) do
				if object.Name == targetCoinName and (object:IsA("BasePart") or object:IsA("Model")) then
					local coinPosition = object:IsA("BasePart") and object.Position or object:GetPivot().Position
					local distance = (hrp.Position - coinPosition).Magnitude

					if distance < shortestDistance then
						shortestDistance = distance
						closestCoin = object
					end
				end
			end

			return closestCoin, shortestDistance
		end

		-- Основной цикл автофарма
		local function farmLoop()
			task.spawn(function()
				while isFarming do
					local character = player.Character or player.CharacterAdded:Wait()
					local hrp = character:FindFirstChild("HumanoidRootPart")

					if hrp then
						local coin, distance = getClosestCoin(character)

						if coin then
							local coinPosition = coin:IsA("BasePart") and coin.Position or coin:GetPivot().Position
							local targetPosition = coinPosition - Vector3.new(0, distanceFromCoin, 0)

							local tweenTime = distance / tweenSpeedStudsPerSec
							if tweenTime <= 0 then tweenTime = 0.05 end

							local tweenInfo = TweenInfo.new(
								tweenTime,
								Enum.EasingStyle.Linear,
								Enum.EasingDirection.Out
							)

							local targetCFrame = CFrame.new(targetPosition, targetPosition + hrp.CFrame.LookVector)

							-- Создаем и запускаем твин
							hrp.CFrame = hrp.CFrame - Vector3.new(0,3,0)
							activeTween = TweenService:Create(hrp, tweenInfo, {CFrame = targetCFrame})
							activeTween:Play()

							-- Ждём завершения полета к монетке
							activeTween.Completed:Wait()
							hrp.CFrame = hrp.CFrame - Vector3.new(0,3,0)
							activeTween = nil
						else
							-- Если монеток нет на карте, ждём 0.5 сек перед следующей проверкой
							task.wait(0.5)
						end
					else
						task.wait(0.5)
					end

					-- Минимальная пауза между циклами для стабильности
					task.wait()
				end
			end)
		end

		-- Функция управления автофармом
		local function autofarm(action)
			if action == "start" then
				if not isFarming then
					isFarming = true
					print("[AutoFarm]: Запущен!")
					farmLoop()
				end
			elseif action == "stop" then
				if isFarming then
					isFarming = false
					print("[AutoFarm]: Остановлен!")

					-- Если персонаж прямо сейчас твинился — сбрасываем движение
					if activeTween then
						activeTween:Cancel()
						activeTween = nil

					end
				end
			end
		end

			if flying then return end
			flying = true

			flyBodyVelocity = Instance.new("BodyVelocity")

			flyBodyVelocity.Parent = rootPart

			flyBodyGyro.Parent = rootPart

			flyConnection = RunService.Heartbeat:Connect(function()
				if not (flyBodyVelocity and flyBodyGyro and rootPart) then return end

				local cf = camera.CFrame
				local dir = Vector3.new()
				local strafe = 0

				if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += cf.LookVector end
				if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= cf.LookVector end
				if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= cf.RightVector; strafe -= 1 end
				if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += cf.RightVector; strafe += 1 end
				if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.new(0, 1, 0) end
				if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) or UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
					dir -= Vector3.new(0, 1, 0)
				end

				flyBodyVelocity.Velocity = dir.Magnitude > 0 and dir.Unit * CONFIG.FlySpeed or Vector3.new()

				-- yaw). Never built from two points, so it can never hit the

				flyBodyGyro.CFrame = CFrame.new(rootPart.Position) * rotationOnly * bank
			end)
		end

			return speedBoosted and CONFIG.BoostedWalkSpeed or defaultWalkSpeed
		end

		local function onHumanoidStateChanged(_, new)
			if not speedGlitching then return end
			if new == Enum.HumanoidStateType.Freefall or new == Enum.HumanoidStateType.Jumping then
				humanoid.WalkSpeed = CONFIG.GlitchWalkSpeed
			elseif new == Enum.HumanoidStateType.Landed or new == Enum.HumanoidStateType.Running or new == Enum.HumanoidStateType.RunningNoPhysics then
				humanoid.WalkSpeed = groundSpeed()
			end
		end

		local function connectHumanoidStateSignal()
			if stateChangedConnection then stateChangedConnection:Disconnect() end
			stateChangedConnection = humanoid.StateChanged:Connect(onHumanoidStateChanged)
		end
		connectHumanoidStateSignal()

		local function applySpeedGlitch(state)
			speedGlitching = state
			humanoid.WalkSpeed = groundSpeed()
		end


		local function teleportAllLocally()
			-- Получаем своего персонажа и точку телепортации
			local Players = game:GetService("Players")
			local localPlayer = Players.LocalPlayer
			local myCharacter = localPlayer.Character
			if not myCharacter or not myCharacter:FindFirstChild("HumanoidRootPart") then 
				return 
			end

			local targetCFrame = myCharacter.HumanoidRootPart.CFrame

			-- Проходим по всем игрокам
			for _, player in pairs(Players:GetPlayers()) do
				-- Пропускаем самого себя
				if player ~= localPlayer and player.Character then
					local hrp = player.Character:FindFirstChild("HumanoidRootPart")
					if hrp then
						-- Смещаем каждого игрока чуть вперед от себя, чтобы они не слипались в одну кучу
						hrp.CFrame = targetCFrame * CFrame.new(0, 0, -3)
					end
				end
			end
		end



		local function teleportNearest(maxDistance)
			local Players = game:GetService("Players")
			local LocalPlayer = Players.LocalPlayer
			maxDistance = maxDistance or 500 -- Радиус поиска по умолчанию

			local myCharacter = LocalPlayer.Character
			if not (myCharacter and myCharacter:FindFirstChild("HumanoidRootPart")) then return end

			local myHrp = myCharacter.HumanoidRootPart
			local nearestPlayer = nil
			local shortestDistance = maxDistance

			-- Поиск ближайшего игрока
			for _, otherPlayer in ipairs(Players:GetPlayers()) do
				if otherPlayer ~= LocalPlayer and otherPlayer.Character and otherPlayer.Character:FindFirstChild("HumanoidRootPart") then
					local otherHrp = otherPlayer.Character.HumanoidRootPart
					local distance = (otherHrp.Position - myHrp.Position).Magnitude

					if distance < shortestDistance then
						shortestDistance = distance
						nearestPlayer = otherPlayer
					end
				end
			end

			-- Телепортируем визуально в 3 студах перед собой
			if nearestPlayer and nearestPlayer.Character and nearestPlayer.Character:FindFirstChild("HumanoidRootPart") then
				local targetHrp = nearestPlayer.Character.HumanoidRootPart
				targetHrp.CFrame = myHrp.CFrame * CFrame.new(0, 0, -3)
			end
		end

		-- abbrev is optional: pass nil to disable the pin-a-shortcut feature
		-- for a particular row (e.g. rarely-needed toggles).
		local function createToggleRow(labelText, initialState, onToggle, abbrev)
			local row = Instance.new("Frame")
			row.Size = UDim2.new(1, 0, 0, 52)
			row.BackgroundColor3 = Color3.fromRGB(21, 17, 18)
			row.Parent = ToggleList

			local rowCorner = Instance.new("UICorner")
			rowCorner.CornerRadius = UDim.new(0, 11)
			rowCorner.Parent = row

			local rowStroke = Instance.new("UIStroke")
			rowStroke.Color = ACCENT
			rowStroke.Transparency = 0.85
			rowStroke.Thickness = 1
			rowStroke.Parent = row

			local label = Instance.new("TextLabel")
			label.Size = UDim2.new(1, -90, 1, 0)
			label.Position = UDim2.new(0, 18, 0, 0)
			label.BackgroundTransparency = 1
			label.Text = labelText
			label.TextColor3 = Color3.fromRGB(235, 225, 225)
			label.TextSize = 15
			label.Font = Enum.Font.GothamMedium
			label.TextXAlignment = Enum.TextXAlignment.Left
			label.Parent = row

			local track = Instance.new("TextButton")
			track.Size = UDim2.new(0, 52, 0, 28)
			track.AnchorPoint = Vector2.new(1, 0.5)
			track.Position = UDim2.new(1, -16, 0.5, 0)
			track.Text = ""
			track.AutoButtonColor = false
			track.BackgroundColor3 = initialState and ACCENT or Color3.fromRGB(45, 38, 39)
			track.Parent = row

			local trackCorner = Instance.new("UICorner")
			trackCorner.CornerRadius = UDim.new(1, 0)
			trackCorner.Parent = track

			local knob = Instance.new("Frame")
			knob.Size = UDim2.new(0, 22, 0, 22)
			knob.AnchorPoint = Vector2.new(0, 0.5)
			knob.Position = initialState and UDim2.new(1, -25, 0.5, 0) or UDim2.new(0, 3, 0.5, 0)
			knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
			knob.Parent = track

			local knobCorner = Instance.new("UICorner")
			knobCorner.CornerRadius = UDim.new(1, 0)
			knobCorner.Parent = knob

			local state = initialState
			local shortcut, shortcutSet = nil, nil

			local function refresh(animate)
				local trackGoal = { BackgroundColor3 = state and ACCENT or Color3.fromRGB(45, 38, 39) }
				local knobGoal = { Position = state and UDim2.new(1, -25, 0.5, 0) or UDim2.new(0, 3, 0.5, 0) }
				if animate then
					TweenService:Create(track, TweenInfo.new(0.15), trackGoal):Play()
					TweenService:Create(knob, TweenInfo.new(0.15), knobGoal):Play()
				else
					track.BackgroundColor3 = trackGoal.BackgroundColor3
					knob.Position = knobGoal.Position
				end
				if shortcutSet then shortcutSet(state) end
			end

			local function doToggle()
				state = not state
				refresh(true)
				onToggle(state)
			end

			local function createOrRemoveShortcut()
				if shortcut then
					shortcut:Destroy()
					shortcut = nil
					shortcutSet = nil
					return
				end

				local bubble = Instance.new("TextButton")
				bubble.Size = UDim2.new(0, 46, 0, 46)
				bubble.BackgroundColor3 = state and ACCENT or Color3.fromRGB(30, 24, 25)
				bubble.AutoButtonColor = false
				bubble.Text = abbrev
				bubble.TextColor3 = Color3.fromRGB(255, 255, 255)
				bubble.TextSize = 14
				bubble.Font = Enum.Font.GothamBold
				bubble.Parent = ShortcutBar

				local bubbleCorner = Instance.new("UICorner")
				bubbleCorner.CornerRadius = UDim.new(1, 0)
				bubbleCorner.Parent = bubble

				local bubbleStroke = Instance.new("UIStroke")
				bubbleStroke.Color = ACCENT
				bubbleStroke.Transparency = 0.4
				bubbleStroke.Thickness = 1.5
				bubbleStroke.Parent = bubble

				-- No dragger on this object -> a plain click event is safe.
				bubble.MouseButton1Click:Connect(doToggle)

				shortcut = bubble
				shortcutSet = function(s)
					bubble.BackgroundColor3 = s and ACCENT or Color3.fromRGB(30, 24, 25)
				end
			end

			-- Short tap = toggle. Long press (0.5s) = pin/unpin shortcut.
			-- Both handled manually via timing so one button can do both.
			local pressStart = nil
			local longPressFired = false

			track.InputBegan:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1
					or input.UserInputType == Enum.UserInputType.Touch then
					pressStart = tick()
					longPressFired = false
					if abbrev then
						task.delay(CONFIG.LongPressTime, function()
							if pressStart and tick() - pressStart >= CONFIG.LongPressTime then
								longPressFired = true
								createOrRemoveShortcut()
							end
						end)
					end
				end
			end)

			track.InputEnded:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1
					or input.UserInputType == Enum.UserInputType.Touch then
					if pressStart and not longPressFired then
						doToggle()
					end
					pressStart = nil
				end
			end)

			row.MouseEnter:Connect(function()
				TweenService:Create(rowStroke, TweenInfo.new(0.15), {Transparency = 0.5}):Play()
			end)
			row.MouseLeave:Connect(function()
				TweenService:Create(rowStroke, TweenInfo.new(0.15), {Transparency = 0.85}):Play()
			end)

			return {
				setState = function(newState)
					state = newState
					refresh(true)
				end,
				getState = function() return state end,
			}
		end

		MenuButton.AnchorPoint = Vector2.new(1, 0.5)
		MenuButton.Position = UDim2.new(1, -20, 0.5, 0)
		MenuButton.BackgroundColor3 = ACCENT
		MenuButton.Text = "="
		MenuButton.TextColor3 = Color3.fromRGB(255, 255, 255)
		MenuButton.TextSize = 20
		MenuButton.Font = Enum.Font.GothamBold
		MenuButton.Parent = gui

		local MenuButtonCorner = Instance.new("UICorner")
		MenuButtonCorner.CornerRadius = UDim.new(1, 0)
		MenuButtonCorner.Parent = MenuButton

		local MenuButtonDragger = Instance.new("UIDragDetector")
		MenuButtonDragger.Parent = MenuButton

		-- This object has both a dragger AND needs a click -> manual
		-- distance-based click detection (dragger on the same object eats
		-- MouseButton1Click otherwise, as covered earlier in this chat).
		do
			local CLICK_THRESHOLD = 6
			local pressStartPos = nil
			MenuButton.InputBegan:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1
					or input.UserInputType == Enum.UserInputType.Touch then
					pressStartPos = input.Position
				end
			end)
			MenuButton.InputEnded:Connect(function(input)
				if not pressStartPos then return end
				if input.UserInputType == Enum.UserInputType.MouseButton1
					or input.UserInputType == Enum.UserInputType.Touch then
					local moved = (input.Position - pressStartPos).Magnitude
					pressStartPos = nil
					if moved < CLICK_THRESHOLD then
						Main.Visible = not Main.Visible
					end
				end
			end)
		end

		-- =====================================================
		--                      HOTKEYS
		-- =====================================================
		UserInputService.InputBegan:Connect(function(input, processed)
			if processed then return end

			if input.KeyCode == CONFIG.FlyKey then
				local newState = not flying
				if newState then startFly() else stopFly() end
				flyToggle.setState(newState)
			elseif input.KeyCode == CONFIG.NoclipKey then
				local newState = not noclipping
				if newState then startNoclip() else stopNoclip() end
				noclipToggle.setState(newState)
			elseif input.KeyCode == CONFIG.ToggleHubKey then
				Main.Visible = not Main.Visible
			end
		end)

		-- =====================================================
		--                    RESPAWN HANDLING
		-- =====================================================
		-- THE bug fix: character/humanoid/rootPart are now actually
		-- reassigned, defaults recaptured, and every connection rebuilt
		-- against the NEW Humanoid instead of the destroyed old one.
		player.CharacterAdded:Connect(function(newChar)
			character = newChar
			humanoid = newChar:WaitForChild("Humanoid")
			rootPart = newChar:WaitForChild("HumanoidRootPart")
			captureDefaults()
			connectHumanoidStateSignal()

			if flyConnection then flyConnection:Disconnect(); flyConnection = nil end
			if noclipConnection then noclipConnection:Disconnect(); noclipConnection = nil end

			flying, noclipping, speedBoosted, highJumping, speedGlitching, classicForced =
				false, false, false, false, false, false

			flyToggle.setState(false)
			noclipToggle.setState(false)
			speedToggle.setState(false)
			glitchToggle.setState(false)
			jumpToggle.setState(false)
			jumpPowerToggle.setState(false)
			-- ESP is left running across respawns on purpose (it watches
			-- OTHER players, your own death doesn't need to reset it).
		end)
		task.wait(0.6)
		keygui:Destroy()
		
	else
		Subtitle.Text = "Incorrect key! try again!"
		Subtitle.TextColor3 = Color3.fromRGB(255, 80, 80)
	end
end
