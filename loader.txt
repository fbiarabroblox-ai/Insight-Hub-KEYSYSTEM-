local Players = game:GetService("Players")

local HITBOX_SIZE = Vector3.new(10, 10, 10)
local HITBOX_TRANSPARENCY = 0.5
local HITBOX_COLOR = Color3.fromRGB(255, 0, 0)
local HITBOX_MATERIAL = Enum.Material.ForceField

local isHitboxEnabled = false
local originalProperties = {}

local function saveOriginalProperties(character)
	if not character or character == localPlayer.Character then return end
	local hrp = character:WaitForChild("HumanoidRootPart", 5)
	if hrp then
		originalProperties[character] = {
			Size = hrp.Size,
			Transparency = hrp.Transparency,
			Color = hrp.Color,
			Material = hrp.Material,
			CanCollide = hrp.CanCollide
		}
	end
end

local function updateHitbox(character)
	-- Исключаем своего персонажа
	if not character or character == localPlayer.Character then return end

	local hrp = character:FindFirstChild("HumanoidRootPart")
	if not hrp then return end

	if isHitboxEnabled then
		if not originalProperties[character] then
			saveOriginalProperties(character)
		end

		hrp.Size = HITBOX_SIZE
		hrp.Transparency = HITBOX_TRANSPARENCY
		hrp.Color = HITBOX_COLOR
		hrp.Material = HITBOX_MATERIAL
		hrp.CanCollide = false
	else
		local original = originalProperties[character]
		if original then
			hrp.Size = original.Size
			hrp.Transparency = original.Transparency
			hrp.Color = original.Color
			hrp.Material = original.Material
			hrp.CanCollide = original.CanCollide
		end
	end
end

local function onPlayerAdded(player)
	-- Пропускаем себя
	if player == localPlayer then return end

	local function setupCharacter(char)
		saveOriginalProperties(char)
		updateHitbox(char)
	end

	if player.Character then
		setupCharacter(player.Character)
	end

	player.CharacterAdded:Connect(setupCharacter)
end

-- Инициализация для других игроков
for _, player in ipairs(Players:GetPlayers()) do
	onPlayerAdded(player)
end
Players.PlayerAdded:Connect(onPlayerAdded)

Players.PlayerRemoving:Connect(function(player)
	if player.Character then
		originalProperties[player.Character] = nil
	end
end)

-- Функция переключения (меняет хитбокс только у ВСЕХ ДРУГИХ игроков)
local function SetHitbox(bool)
	if typeof(bool) ~= "boolean" then return end
	isHitboxEnabled = bool

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= localPlayer and player.Character then
			updateHitbox(player.Character)
		end
	end
end

		function Timer:Start(onUpdateCallback)
			if self._running then return end
			self._running = true

    	   return true
        end

    local function checkObject(object)
    	if processed[object] then return end

    	local name = object.Name:lower()
    	if name ~= "gun" and name ~= "knife" then return end

	    local parent = object.Parent
    	if not parent then return end
    	if processedParents[name][parent] then return end

    	processed[object] = true
    	processedParents[name][parent] = true

    	if name == "gun" then
	    	onGunFound(object)
    	else
    		onKnifeFound(object)
    	end
    end

		local targetCoinName = "Coin_Server"
		local tweenSpeedStudsPerSec = 16
		local UNDER_OFFSET = 3   -- на сколько стадов ниже монетки летим и куда возвращаемся
		local TOUCH_TIME = 0.5  -- ожидание в монетке
		local SKIP_TIME = 10     -- сколько сек не возвращаемся к монетке, если она не исчезла

		local isFarming = false
		local activeTween = nil
		local farmHold = nil
		local visited = {} -- [coin] = время посещения

		-- Чистка памяти: удаляем удалённые монетки и просроченные записи
		local function cleanVisited()
			local now = os.clock()
			for coin, t in pairs(visited) do
				if not coin.Parent or now - t > SKIP_TIME then
					visited[coin] = nil
				end
			end
		end

		local function getClosestCoin(character)
			local hrp = character:FindFirstChild("HumanoidRootPart")
			if not hrp then return nil end

			local closestCoin = nil
			local shortestDistance = math.huge
			local needle = string.lower(tostring(targetCoinName))

			for _, object in ipairs(Workspace:GetDescendants()) do
				if not visited[object]
					and (object:IsA("BasePart") or object:IsA("Model"))
					and string.find(string.lower(object.Name), needle, 1, true) then

					local pos = object:IsA("BasePart") and object.Position or object:GetPivot().Position
					local distance = (hrp.Position - pos).Magnitude

					if distance < shortestDistance then
						shortestDistance = distance
						closestCoin = object
					end
				end
			end

			return closestCoin, shortestDistance
		end

		local function freezeCharacter(character)
			local hrp = character:FindFirstChild("HumanoidRootPart")
			local hum = character:FindFirstChildOfClass("Humanoid")
			if not (hrp and hum) then return end

			hum.PlatformStand = true
			if not (farmHold and farmHold.Parent) then
				farmHold = Instance.new("BodyVelocity")
				farmHold.MaxForce = Vector3.new(1, 1, 1) * math.huge
				farmHold.Velocity = Vector3.zero
				farmHold.Parent = hrp
			end
		end

		local function unfreezeCharacter()
			if farmHold then
				farmHold:Destroy()
				farmHold = nil
			end
			local character = player.Character
			local hum = character and character:FindFirstChildOfClass("Humanoid")
			if hum then hum.PlatformStand = false end
		end

		local function flyTo(hrp, position, rot)
			local distance = (position - hrp.Position).Magnitude
			local time = math.max(distance / tweenSpeedStudsPerSec, 0.05)

			activeTween = TweenService:Create(
				hrp,
				TweenInfo.new(time, Enum.EasingStyle.Linear, Enum.EasingDirection.Out),
				{CFrame = CFrame.new(position) * rot}
			)
			activeTween:Play()
			activeTween.Completed:Wait()
			activeTween = nil
		end

		local function teleportTo(hrp, position, rot)
			hrp.AssemblyLinearVelocity = Vector3.zero
			hrp.CFrame = CFrame.new(position) * rot
		end

		local function farmLoop()
			task.spawn(function()
				while isFarming do
					local character = player.Character or player.CharacterAdded:Wait()
					local hrp = character:FindFirstChild("HumanoidRootPart")

					if hrp then
						freezeCharacter(character)
						cleanVisited()
						local coin = getClosestCoin(character)

						if coin and coin.Parent then
							local coinPos = coin:IsA("BasePart") and coin.Position or coin:GetPivot().Position
							local underY = coinPos.Y - UNDER_OFFSET
							local rot = hrp.CFrame.Rotation

							-- 1) Вниз под монетку (на 3 стада ниже её высоты)
							local cur = hrp.Position
							teleportTo(hrp, Vector3.new(cur.X, underY, cur.Z), rot)
							if not isFarming then break end

							-- 2) Под картой к точке под монеткой
							flyTo(hrp, Vector3.new(coinPos.X, underY, coinPos.Z), rot)
							if not isFarming then break end

							-- 3) Телепорт в монетку и ожидание
							if coin.Parent then
								teleportTo(hrp, coinPos, rot)
								task.wait(TOUCH_TIME)
								teleportTo(hrp, Vector3.new(coinPos.X, underY, coinPos.Z), rot)
							end

							-- запоминаем монетку, чтобы не вернуться к ней
							visited[coin] = os.clock()
							if not isFarming then break end

						else
							task.wait(0.5)
						end
					else
						task.wait(0.5)
					end

					task.wait()
				end

				unfreezeCharacter()
			end)
		end

		local function autofarm(action)
			if action == "start" then
				if not isFarming then
					isFarming = true
					visited = {}
					print("[AutoFarm]: Запущен!")
					farmLoop()
				end
			elseif action == "stop" then
				if isFarming then
					isFarming = false
					print("[AutoFarm]: Остановлен!")

					if activeTween then
						activeTween:Cancel()
						activeTween = nil
					end
					unfreezeCharacter()
					visited = {}
				end
			end
		end

		local function startautofarm(state)

				autofarm("stop")
			end
		end

		local function startFly()
			if flying then return end

			flyBodyVelocity = Instance.new("BodyVelocity")
			flyBodyVelocity.MaxForce = Vector3.new(1, 1, 1) * math.huge

			flyBodyVelocity.Parent = rootPart

			flyBodyGyro = Instance.new("BodyGyro")
			flyBodyGyro.MaxTorque = Vector3.new(1, 1, 1) * math.huge
			flyBodyGyro.P = 3000
			flyBodyGyro.CFrame = rootPart.CFrame
			flyBodyGyro.Parent = rootPart

			-- Кнопка прыжка (телефон и Space на ПК) даёт сигнал JumpRequest
			jumpConn = UserInputService.JumpRequest:Connect(function()
				lastJumpRequest = os.clock()
			end)

			flyConnection = RunService.Heartbeat:Connect(function()
				if not (flyBodyVelocity and flyBodyGyro and rootPart) then return end

				local cf = camera.CFrame
				local dir = Vector3.new()
				local strafe = 0

				-- Джойстик / WASD: x = вправо-влево, z = назад-вперёд (вперёд отрицательный)
				local mv = Vector3.new()
				if controls then
					mv = controls:GetMoveVector()
				else
					local md = humanoid.MoveDirection
					if md.Magnitude > 0 then
						mv = Vector3.new(cf.RightVector:Dot(md), 0, -cf.LookVector:Dot(md))
					end
				end

				if mv.Magnitude > 0 then
					dir += cf.LookVector * -mv.Z + cf.RightVector * mv.X
					strafe = mv.X
				end

				-- Прыжок = подъём вверх (зажатая кнопка или недавнее нажатие)
				local jumping = humanoid.Jump or (os.clock() - lastJumpRequest) < 0.2
				if jumping or UserInputService:IsKeyDown(Enum.KeyCode.Space) then
					dir += Vector3.new(0, 1, 0)
				end

				-- Спуск на ПК
				if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift)
					or UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
					dir -= Vector3.new(0, 1, 0)
				end

				flyBodyVelocity.Velocity = dir.Magnitude > 0 and dir.Unit * CONFIG.FlySpeed or Vector3.new()

				local rotationOnly = cf - cf.Position
				local bank = CFrame.Angles(0, 0, -strafe * CONFIG.FlyBankAngle)
				flyBodyGyro.CFrame = CFrame.new(rootPart.Position) * rotationOnly * bank
			end)
		end

		local function teleportNearest(maxDistance)

			local myCharacter = LocalPlayer.Character

			local nearestPlayer = nil

			-- Телепортируем визуально в 3 студах перед собой
			if nearestPlayer and nearestPlayer.Character and nearestPlayer.Character:FindFirstChild("HumanoidRootPart") then
				local targetHrp = nearestPlayer.Character.HumanoidRootPart
				targetHrp.CFrame = myHrp.CFrame * CFrame.new(0, 0, -3)

        local AIM_SMOOTH = 1 -- 1 = мгновенный доворот, меньше = плавнее

        local function toolMatches(char, keywords)
        	for _, child in ipairs(char:GetChildren()) do
        		if child:IsA("Tool") then
	        		local n = child.Name:lower()
		        	for _, kw in ipairs(keywords) do
	    		    	if n:find(kw, 1, true) then return true end
    		    	end
    		    end
    	    end
        	return false
         end

         local function isAlive(char)
        	local hum = char and char:FindFirstChildOfClass("Humanoid")
        	return hum ~= nil and hum.Health > 0 and char:FindFirstChild("HumanoidRootPart") ~= nil
         end

        local function getAimTarget()
        	local myChar = player.Character
        	if not isAlive(myChar) then return nil end
        	local myPos = myChar.HumanoidRootPart.Position

        	local onlyKnifeHolders
        	if toolMatches(myChar, CONFIG.ESP.KnifeKeywords) then
        		onlyKnifeHolders = false -- у тебя нож: целимся в ближайшего
        	elseif toolMatches(myChar, CONFIG.ESP.GunKeywords) then
        		onlyKnifeHolders = true  -- у тебя пистолет: целимся в игрока с ножом
        	else
        		return nil
        	end

        	local best, bestDist = nil, math.huge
        	for _, plr in ipairs(Players:GetPlayers()) do
        		if plr ~= player and isAlive(plr.Character) then
	        		if not onlyKnifeHolders or toolMatches(plr.Character, CONFIG.ESP.KnifeKeywords) then
		        		local hrp = plr.Character.HumanoidRootPart
		        		local d = (hrp.Position - myPos).Magnitude
	        			if d < bestDist then
			        		best, bestDist = hrp, d
	     		    	end
		        	end
    	    	end
        	end
        	return best
       end

       local function aimStep()
        	local target = getAimTarget()
        	if not target then return end
        	local cam = workspace.CurrentCamera
        	local goal = CFrame.lookAt(cam.CFrame.Position, target.Position)
        	cam.CFrame = cam.CFrame:Lerp(goal, AIM_SMOOTH)
        end

        	end

		-- ---------- Toggle row constructor (with long-press shortcuts) ----------
		-- abbrev is optional: pass nil to disable the pin-a-shortcut feature
		-- for a particular row (e.g. rarely-needed toggles).
		local function createToggleRow(labelText, initialState, onToggle, abbrev)
			local row = Instance.new("Frame")
			row.Size = UDim2.new(1, 0, 0, 52)
			row.BackgroundColor3 = Color3.fromRGB(21, 17, 18)
			row.Parent = ToggleList

			local rowCorner = Instance.new("UICorner")
			rowCorner.CornerRadius = UDim.new(0, 11)
			rowCorner.Parent = row

			local rowStroke = Instance.new("UIStroke")
			rowStroke.Color = ACCENT
			rowStroke.Transparency = 0.85
			rowStroke.Thickness = 1
			rowStroke.Parent = row

			local label = Instance.new("TextLabel")
			label.Size = UDim2.new(1, -90, 1, 0)
			label.Position = UDim2.new(0, 18, 0, 0)
			label.BackgroundTransparency = 1
			label.Text = labelText
			label.TextColor3 = Color3.fromRGB(235, 225, 225)
			label.TextSize = 15
			label.Font = Enum.Font.GothamMedium
			label.TextXAlignment = Enum.TextXAlignment.Left
			label.Parent = row

			local track = Instance.new("TextButton")
			track.Size = UDim2.new(0, 52, 0, 28)
			track.AnchorPoint = Vector2.new(1, 0.5)
			track.Position = UDim2.new(1, -16, 0.5, 0)
			track.Text = ""
			track.AutoButtonColor = false
			track.BackgroundColor3 = initialState and ACCENT or Color3.fromRGB(45, 38, 39)
			track.Parent = row

			local trackCorner = Instance.new("UICorner")
			trackCorner.CornerRadius = UDim.new(1, 0)
			trackCorner.Parent = track

			local knob = Instance.new("Frame")
			knob.Size = UDim2.new(0, 22, 0, 22)
			knob.AnchorPoint = Vector2.new(0, 0.5)
			knob.Position = initialState and UDim2.new(1, -25, 0.5, 0) or UDim2.new(0, 3, 0.5, 0)
			knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
			knob.Parent = track

			local knobCorner = Instance.new("UICorner")
			knobCorner.CornerRadius = UDim.new(1, 0)
			knobCorner.Parent = knob

			local state = initialState
			local shortcut, shortcutSet = nil, nil

			local function refresh(animate)
				local trackGoal = { BackgroundColor3 = state and ACCENT or Color3.fromRGB(45, 38, 39) }
				local knobGoal = { Position = state and UDim2.new(1, -25, 0.5, 0) or UDim2.new(0, 3, 0.5, 0) }
				if animate then
					TweenService:Create(track, TweenInfo.new(0.15), trackGoal):Play()
					TweenService:Create(knob, TweenInfo.new(0.15), knobGoal):Play()
				else
					track.BackgroundColor3 = trackGoal.BackgroundColor3
					knob.Position = knobGoal.Position
				end
				if shortcutSet then shortcutSet(state) end
			end

			local function doToggle()
				state = not state
				refresh(true)
				onToggle(state)
			end

			local function createOrRemoveShortcut()
				if shortcut then
					shortcut:Destroy()
					shortcut = nil
					shortcutSet = nil
					return
				end

				local bubble = Instance.new("TextButton")
				bubble.Size = UDim2.new(0, 46, 0, 46)
				bubble.BackgroundColor3 = state and ACCENT or Color3.fromRGB(30, 24, 25)
				bubble.AutoButtonColor = false
				bubble.Text = abbrev
				bubble.TextColor3 = Color3.fromRGB(255, 255, 255)
				bubble.TextSize = 14
				bubble.Font = Enum.Font.GothamBold
				bubble.Parent = ShortcutBar

				local bubbleCorner = Instance.new("UICorner")
				bubbleCorner.CornerRadius = UDim.new(1, 0)
				bubbleCorner.Parent = bubble

				local bubbleStroke = Instance.new("UIStroke")
				bubbleStroke.Color = ACCENT
				bubbleStroke.Transparency = 0.4
				bubbleStroke.Thickness = 1.5
				bubbleStroke.Parent = bubble

				-- No dragger on this object -> a plain click event is safe.
				bubble.MouseButton1Click:Connect(doToggle)

				shortcut = bubble
				shortcutSet = function(s)
					bubble.BackgroundColor3 = s and ACCENT or Color3.fromRGB(30, 24, 25)
				end
			end

			-- Short tap = toggle. Long press (0.5s) = pin/unpin shortcut.
			-- Both handled manually via timing so one button can do both.
			local pressStart = nil
			local longPressFired = false

			track.InputBegan:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1
					or input.UserInputType == Enum.UserInputType.Touch then
					pressStart = tick()
					longPressFired = false
					if abbrev then
						task.delay(CONFIG.LongPressTime, function()
							if pressStart and tick() - pressStart >= CONFIG.LongPressTime then
								longPressFired = true
								createOrRemoveShortcut()
							end
						end)
					end
				end
			end)

			track.InputEnded:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1
					or input.UserInputType == Enum.UserInputType.Touch then
					if pressStart and not longPressFired then
						doToggle()
					end
					pressStart = nil
				end
			end)

			row.MouseEnter:Connect(function()
				TweenService:Create(rowStroke, TweenInfo.new(0.15), {Transparency = 0.5}):Play()
			end)
			row.MouseLeave:Connect(function()
				TweenService:Create(rowStroke, TweenInfo.new(0.15), {Transparency = 0.85}):Play()
			end)

			return {
				setState = function(newState)
					state = newState
					refresh(true)
				end,
				getState = function() return state end,
			}
		end

			end)

			-- =====================================================
			--     PERSISTENT MENU BUTTON (open/close hub — mobile)
			-- =====================================================
			local MenuButton = Instance.new("TextButton")
			MenuButton.Size = UDim2.new(0, 46, 0, 46)
			MenuButton.AnchorPoint = Vector2.new(1, 0.5)
			MenuButton.Position = UDim2.new(1, -20, 0.5, 0)
			MenuButton.BackgroundColor3 = ACCENT
			MenuButton.Text = "="
			MenuButton.TextColor3 = Color3.fromRGB(255, 255, 255)
			MenuButton.TextSize = 20
			MenuButton.Font = Enum.Font.GothamBold
			MenuButton.Parent = gui

			local MenuButtonCorner = Instance.new("UICorner")
			MenuButtonCorner.CornerRadius = UDim.new(1, 0)
			MenuButtonCorner.Parent = MenuButton

			local MenuButtonDragger = Instance.new("UIDragDetector")
			MenuButtonDragger.Parent = MenuButton

			-- This object has both a dragger AND needs a click -> manual
			-- distance-based click detection (dragger on the same object eats
			-- MouseButton1Click otherwise, as covered earlier in this chat).
			do
				local CLICK_THRESHOLD = 6
				local pressStartPos = nil
				MenuButton.InputBegan:Connect(function(input)
					if input.UserInputType == Enum.UserInputType.MouseButton1
						or input.UserInputType == Enum.UserInputType.Touch then
						pressStartPos = input.Position
					end
				end)
				MenuButton.InputEnded:Connect(function(input)
					if not pressStartPos then return end
					if input.UserInputType == Enum.UserInputType.MouseButton1
						or input.UserInputType == Enum.UserInputType.Touch then
						local moved = (input.Position - pressStartPos).Magnitude
						pressStartPos = nil
						if moved < CLICK_THRESHOLD then
							Main.Visible = not Main.Visible
						end
					end
				end)
			end

			-- =====================================================
			--                      HOTKEYS
			-- =====================================================
			UserInputService.InputBegan:Connect(function(input, processed)
				if processed then return end

				if input.KeyCode == CONFIG.FlyKey then
					local newState = not flying
					if newState then startFly() else stopFly() end
					flyToggle.setState(newState)
				elseif input.KeyCode == CONFIG.NoclipKey then
					local newState = not noclipping
					if newState then startNoclip() else stopNoclip() end
					noclipToggle.setState(newState)
				elseif input.KeyCode == CONFIG.ToggleHubKey then
					Main.Visible = not Main.Visible
				end
			end)

			-- =====================================================
			--                    RESPAWN HANDLING
			-- =====================================================
			-- THE bug fix: character/humanoid/rootPart are now actually
			-- reassigned, defaults recaptured, and every connection rebuilt
			-- against the NEW Humanoid instead of the destroyed old one.
			player.CharacterAdded:Connect(function(newChar)
				character = newChar
				humanoid = newChar:WaitForChild("Humanoid")
				rootPart = newChar:WaitForChild("HumanoidRootPart")
				captureDefaults()
				connectHumanoidStateSignal()

				if flyConnection then flyConnection:Disconnect(); flyConnection = nil end
				if noclipConnection then noclipConnection:Disconnect(); noclipConnection = nil end

				flying, noclipping, speedBoosted, highJumping, speedGlitching, classicForced =
					false, false, false, false, false, false

				flyToggle.setState(false)
				noclipToggle.setState(false)
				speedToggle.setState(false)
				glitchToggle.setState(false)
				jumpToggle.setState(false)
				jumpPowerToggle.setState(false)
				-- ESP is left running across respawns on purpose (it watches
				-- OTHER players, your own death doesn't need to reset it).
                resetAll()
			end)

			local finalTime = myTimer:Stop()
			showNotification("Insight Hub loaded in: ".. timet)

			task.wait(0.6)
			keygui:Destroy()


	else
		Subtitle.Text = "Incorrect key! try again!"
		Subtitle.TextColor3 = Color3.fromRGB(255, 80, 80)
	end
end
